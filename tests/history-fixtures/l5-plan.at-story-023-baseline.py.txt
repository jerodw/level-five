#!/usr/bin/env python3
"""Plan a story interactively with the planner agent.

Usage: l5-plan "<story request>"
Starts an interactive Claude Code session carrying the planner prompt.
The planner interviews you, iterates on the plan, and writes the approved
story artifact to .harness/stories/.
"""
import os
import sys
from pathlib import Path

HARNESS_ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(HARNESS_ROOT / "orchestration"))

import context_assembler  # noqa: E402
import harness_config  # noqa: E402


def main() -> None:
    if len(sys.argv) < 2:
        sys.exit(__doc__.strip())
    request = " ".join(sys.argv[1:])
    target_root = harness_config.find_target_root(Path.cwd())
    config = harness_config.load_config(target_root)
    workflow = harness_config.load_workflow(
        HARNESS_ROOT, config.get("workflow", "story-workflow")
    )
    rules = harness_config.load_rules(HARNESS_ROOT)
    # The planner is not a workflow stage, so no coordinator renders its
    # template. Render it here against the same schema and workflow context
    # the coordinator builds from, so the story contract and the workflow's
    # stage rules reach the planner as the enforced files themselves rather
    # than as copies of them that can drift.
    template = context_assembler.load_template(HARNESS_ROOT, "planner.md")
    context = context_assembler.schema_context(HARNESS_ROOT)
    context.update(context_assembler.workflow_context(workflow, rules))
    planner_prompt = context_assembler.render(template, context)
    os.execvp(
        "claude",
        [
            "claude",
            "--permission-mode",
            "acceptEdits",
            "--append-system-prompt",
            planner_prompt,
            f"Story request: {request}",
        ],
    )


if __name__ == "__main__":
    main()
